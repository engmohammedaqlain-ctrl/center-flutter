/// ورقة المعلمون والمواد — تظهر عند الطلب فقط، مع زر تعديل لمن يملك الصلاحية.
Future<void> _showSubjectTeachersSheet(
  BuildContext context, {
  required Classroom room,
  required bool canEdit,
}) async {
  final store = StoreScope.of(context);
  await showModalBottomSheet<void>(
    context: context,
    backgroundColor: Colors.white,
    isScrollControlled: true,
    shape: const RoundedRectangleBorder(
      borderRadius: BorderRadius.vertical(top: Radius.circular(Corner.dialog)),
    ),
    builder: (ctx) {
      final assigned = store.sectionSubjectGroups(room.id).where((g) => g.subjectId.isNotEmpty).toList()
        ..sort((a, b) => store.subjectName(a.subjectId).compareTo(store.subjectName(b.subjectId)));
      return SafeArea(
        child: Padding(
          padding: const EdgeInsets.fromLTRB(16, 12, 16, 16),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              Text(
                'المعلمون والمواد',
                style: TextStyle(
                  fontFamily: AppText.family,
                  fontWeight: FontWeight.w800,
                  fontSize: 15,
                  color: AppColors.heading,
                ),
              ),
              const SizedBox(height: 4),
              Text(
                room.name,
                style: const TextStyle(color: AppColors.muted, fontSize: 12.5),
              ),
              const SizedBox(height: 12),
              if (assigned.isEmpty)
                const Padding(
                  padding: EdgeInsets.symmetric(vertical: 20),
                  child: Text('لا مواد مسندة بعد', textAlign: TextAlign.center, style: TextStyle(color: AppColors.muted)),
                )
              else
                ConstrainedBox(
                  constraints: BoxConstraints(maxHeight: MediaQuery.sizeOf(ctx).height * 0.45),
                  child: ListView.separated(
                    shrinkWrap: true,
                    itemCount: assigned.length,
                    separatorBuilder: (_, __) => const Divider(height: 1),
                    itemBuilder: (_, i) {
                      final g = assigned[i];
                      return ListTile(
                        dense: true,
                        contentPadding: EdgeInsets.zero,
                        title: Text(
                          store.subjectName(g.subjectId),
                          style: const TextStyle(fontWeight: FontWeight.w700, fontSize: 13.5),
                        ),
                        subtitle: Text(
                          g.teacherId.trim().isEmpty ? 'غير مسند' : store.teacherName(g.teacherId),
                          style: const TextStyle(fontSize: 12, color: AppColors.muted),
                        ),
                      );
                    },
                  ),
                ),
              if (canEdit) ...[
                const SizedBox(height: 12),
                PrimaryButton(
                  label: 'تعديل معلمي المواد',
                  icon: Icons.edit_outlined,
                  expand: true,
                  onPressed: () {
                    Navigator.pop(ctx);
                    _assignSubjectTeachers(context, room);
                  },
                ),
              ],
            ],
          ),
        ),
      );
    },
  );
}
